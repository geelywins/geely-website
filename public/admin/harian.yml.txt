name: Posting artikel baru ke media sosial

on:
  workflow_dispatch:
  # Setiap hari pukul 07:00 WIB (setelah website dibangun ulang pukul 05:30)
  schedule:
    - cron: '0 0 * * *'

permissions:
  contents: write

concurrency:
  group: posting
  cancel-in-progress: false

jobs:
  posting:
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
      - name: Kirim artikel baru
        env:
          N8N_WEBHOOK_URL: ${{ secrets.N8N_WEBHOOK_URL }}
          TELEGRAM_BOT_TOKEN: ${{ secrets.TELEGRAM_BOT_TOKEN }}
          TELEGRAM_CHAT_ID: ${{ secrets.TELEGRAM_CHAT_ID }}
        run: node scripts/posting.mjs
      - name: Simpan catatan
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add src/data/dipost.json
          if git diff --cached --quiet; then echo "Tidak ada perubahan"; exit 0; fi
          git commit -m "Catat artikel yang sudah diposting"
          git pull --rebase origin "${{ github.ref_name }}" || true
          git push origin "HEAD:${{ github.ref_name }}"
