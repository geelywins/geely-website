name: Laporan mingguan (Google, link, kecepatan)

on:
  workflow_dispatch:
  # Setiap Senin pukul 06:00 WIB
  schedule:
    - cron: '0 23 * * 0'

permissions:
  contents: write

concurrency:
  group: laporan
  cancel-in-progress: false

jobs:
  laporan:
    runs-on: ubuntu-latest
    timeout-minutes: 40
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
      - name: Buat laporan
        env:
          GSC_KEY: ${{ secrets.GSC_KEY }}
        run: node scripts/laporan.mjs
      - name: Simpan laporan
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add src/data/laporan.json
          if git diff --cached --quiet; then echo "Tidak ada perubahan"; exit 0; fi
          git commit -m "Laporan mingguan otomatis"
          git pull --rebase origin "${{ github.ref_name }}" || true
          git push origin "HEAD:${{ github.ref_name }}"
